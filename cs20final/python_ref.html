Python 3 Quick Reference

Variables

x = 5
name = "Alice"
is_valid = True

Type Conversion (Casting)

# Convert to integer
num1 = int("10")    # 10
num2 = int(5.7)     # 5 (truncates decimal)

# Convert to float
pi = float("3.14")  # 3.14
val = float(8)      # 8.0

# Convert to string
text = str(100)     # "100"

User Input

# Reads input as a string
user_text = input("Enter text: ")

# Converts input to an integer for calculations
user_number = int(input("Enter a number: "))

Basic Math

addition = 5 + 3
subtraction = 5 - 3
multiplication = 5 * 3
division = 5 / 3       # Returns float (1.666...)
floor_division = 5 // 3 # Returns int (1)
modulus = 5 % 3        # Remainder (2)
exponent = 5 ** 3      # Power (125)

Functions

def my_function(parameter1, parameter2):
    result = parameter1 + parameter2
    return result

# Calling a function
value = my_function(10, 20)

For Loops

# Loop from 0 up to (but not including) 5
for i in range(5):
    print(i)

# Loop through list items
for item in my_list:
    print(item)

While Loops

count = 0
while count < 5:
    print(count)
    count = count + 1

Lists

# Create a list
numbers = [10, 20, 30, 40]

# Access item (indexing starts at 0)
first_item = numbers[0]

# Add item to the end
numbers.append(50)

# Get length of list
total_items = len(numbers)

Slicing (Lists & Strings)

# syntax -> element[start:stop] (stop index is excluded)
items = [0, 1, 2, 3, 4, 5]
text = "Python"

# Get elements from index 1 up to 4
sub_list = items[1:4]  # [1, 2, 3]
sub_text = text[1:4]   # "yth"

# From start up to index 3
start_to_three = items[:3]  # [0, 1, 2]

# From index 2 to the end
two_to_end = items[2:]  # [2, 3, 4, 5]
